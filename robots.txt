# 색인 대상은 랜딩 한 장뿐이다.
# search·login·mypage는 각 파일의 <meta name="robots" content="noindex">가 막는다 —
# 셋 다 상태에 따라 내용이 달라지는 화면이라 색인할 고정 콘텐츠가 없다.
#
# 여기서 Disallow로 막지 않는 이유: Disallow는 크롤링을 막을 뿐 색인을 막지 않는다.
# 오히려 막으면 크롤러가 noindex를 읽으러 들어오지도 못해서, 외부 링크만으로
# 주소가 색인될 수 있다. 크롤링은 열어 두고 페이지가 직접 noindex를 말하게 한다.

User-agent: *
Allow: /

# API는 크롤링할 것이 없다. POST 전용이고 GET은 405를 낸다.
Disallow: /api/

Sitemap: https://food-voyage-puce.vercel.app/sitemap.xml
